class_name CameraRig
extends Camera2D
## CÃ¢mera com dois modos + tremor de tela.
##  STATIC: parada no centro da arena (combates em arena fechada).
##  FOLLOW: segue o alvo suavemente, limitada pelo retÃ¢ngulo da arena.

enum Mode { STATIC, FOLLOW }

@export var mode: Mode = Mode.STATIC
@export var target: Node2D
## RetÃ¢ngulo da arena em coordenadas do mundo. Vira o limite da cÃ¢mera.
@export var arena_rect: Rect2 = Rect2(0, 0, 640, 360)
@export var follow_speed: float = 8.0
@export var shake_decay: float = 30.0

var _shake: float = 0.0


func _ready() -> void:
	limit_left = int(arena_rect.position.x)
	limit_top = int(arena_rect.position.y)
	limit_right = int(arena_rect.end.x)
	limit_bottom = int(arena_rect.end.y)
	if mode == Mode.STATIC:
		global_position = arena_rect.get_center()
	elif target:
		global_position = target.global_position


func _process(delta: float) -> void:
	if mode == Mode.FOLLOW and target:
		var t := 1.0 - exp(-follow_speed * delta)  # suavizaÃ§Ã£o independente de FPS
		global_position = global_position.lerp(target.global_position, t)

	if _shake > 0.0:
		offset = Vector2(randf_range(-1.0, 1.0), randf_range(-1.0, 1.0)) * _shake
		_shake = move_toward(_shake, 0.0, shake_decay * delta)
	else:
		offset = Vector2.ZERO


## Chame ao acertar um golpe forte, tomar dano, etc. (strength em pixels)
func shake(strength: float = 3.0) -> void:
	_shake = maxf(_shake, strength)

class_name AmmoComponent
extends Node
## MuniÃ§Ã£o contada (pedras). SÃ³ guarda e controla a quantidade; quem atira e quem mostra
## na tela (HUD) apenas conversam com este componente.
## Quando existir o RunState, SÃ“ este arquivo muda para ler/gravar a muniÃ§Ã£o lÃ¡,
## para ela persistir entre salas. Quem usa o componente nÃ£o precisa mudar.

## Emitido sempre que a quantidade muda (o HUD escuta este sinal).
signal changed(current: int, maximum: int)
## Emitido ao tentar gastar sem ter muniÃ§Ã£o (bom lugar para um som de "vazio").
signal depleted

@export var max_ammo: int = 6

var current: int = 0


func _ready() -> void:
	current = max_ammo


func has(amount: int = 1) -> bool:
	return current >= amount


## Gasta muniÃ§Ã£o. Retorna false (e emite `depleted`) se nÃ£o houver o suficiente.
func consume(amount: int = 1) -> bool:
	if current < amount:
		depleted.emit()
		return false
	current -= amount
	changed.emit(current, max_ammo)
	return true


## Recarrega. Sem argumento, enche tudo (use na fogueira e na loja).
func refill(amount: int = -1) -> void:
	var target: int = max_ammo if amount < 0 else mini(current + amount, max_ammo)
	if target == current:
		return
	current = target
	changed.emit(current, max_ammo)


## Para relÃ­quias que aumentam a capacidade no futuro.
func set_max(value: int, refill_to_full: bool = false) -> void:
	max_ammo = maxi(value, 0)
	current = max_ammo if refill_to_full else mini(current, max_ammo)
	changed.emit(current, max_ammo)

class_name AttackComponent
extends Node2D

## Quem decide QUANDO atacar Ã© o dono (ex.: o estado Attack do Player), chamando start().
## Este componente sÃ³ executa as fases, lendo tempos e dano do AttackData.
## Ã‰ um Node2D (e nÃ£o Node) para o Hitbox filho herdar posiÃ§Ã£o/flip do Valtair.
## O pai precisa ser um CharacterBody2D e deve existir um filho Hitbox chamado "Hitbox".

signal active_ended

enum State { IDLE, STARTUP, ACTIVE, RECOVERY }

@export var attack_data: AttackData
@export var timer_driven: bool = true
## Efeito visual do golpe (ex.: o nÃ³ SlashFX). Fica escondido e sÃ³ aparece na fase ACTIVE.
@export var vfx: CanvasItem

@onready var body: CharacterBody2D = get_parent() as CharacterBody2D
@onready var hitbox: Hitbox = $Hitbox as Hitbox

func _ready() -> void:
	assert(attack_data != null, "AttackComponent precisa de um AttackData atribuÃ­do no Inspetor!")
	hitbox.source = body
	hitbox.damage = attack_data.damage
	hitbox.hit_landed.connect(func(h: Hurtbox) -> void: hit_landed.emit(h))
	_set_vfx_visible(false)


func is_attacking() -> bool:
	return state != State.IDLE


func set_facing(facing: int) -> void:
	scale.x = float(facing)


## Inicia o golpe. Ignorado se jÃ¡ estiver atacando.
func start() -> void:
	if state != State.IDLE:
		return
	state = State.STARTUP
	_timer = attack_data.startup_time()
	attack_started.emit()
	if timer_driven and attack_data.startup_frames <= 0:
		enter_active()


## Deve ser chamado a cada frame de fÃ­sica por quem Ã© dono do ataque.
func update(delta: float) -> void:
	if state == State.IDLE or not timer_driven:
		return
	_timer -= delta
	if _timer > 0.0:
		return
	match state:
		State.STARTUP: enter_active()
		State.ACTIVE: enter_recovery()
		State.RECOVERY: finish()


## Interrompe o golpe (ex.: o dono saiu do estado). Seguro de chamar em qualquer momento.
func cancel() -> void:
	if state == State.IDLE:
		return
	finish()


## --- PÃºblicos: o AnimationPlayer pode chamar estes trÃªs ---
func enter_active() -> void:
	state = State.ACTIVE
	_timer = attack_data.active_time()
	hitbox.damage = attack_data.damage
	hitbox.activate()
	_set_vfx_visible(true)
	active_started.emit()


func enter_recovery() -> void:
	state = State.RECOVERY
	_timer = attack_data.recovery_time()
	hitbox.deactivate()
	_set_vfx_visible(false)
	active_ended.emit()


func finish() -> void:
	state = State.IDLE
	hitbox.deactivate()
	_set_vfx_visible(false)
	attack_finished.emit()


func _set_vfx_visible(value: bool) -> void:
	if vfx != null:
		vfx.visible = value

	health = max_health

class_name JumpComponent
extends Node
## Movimento VERTICAL: gravidade + pulo. O pai precisa ser um CharacterBody2D.
## Cada "truque" de game feel pode ser desligado pelo Inspector:
##  - altura variÃ¡vel: variable_height = false
##  - coyote time:     coyote_time = 0
##  - jump buffer:     jump_buffer_time = 0
##  - pulo duplo:      max_jumps = 2

signal jumped
signal landed

@export var jump_velocity: float = -340.0  # negativo = para cima (Y cresce para baixo)
@export var gravity: float = 900.0
## Cair mais rÃ¡pido do que subir deixa o pulo menos "flutuante".
@export var fall_gravity_multiplier: float = 1.5
@export var max_fall_speed: float = 600.0
@export_group("Game feel")
@export var variable_height: bool = true
## Ao soltar o botÃ£o subindo, a velocidade vertical Ã© multiplicada por isso.
@export_range(0.0, 1.0) var jump_cut_multiplier: float = 0.4
@export var coyote_time: float = 0.10
@export var jump_buffer_time: float = 0.10
@export var max_jumps: int = 1

@onready var body: CharacterBody2D = get_parent() as CharacterBody2D

var _coyote: float = 0.0
var _buffer: float = 0.0
var _air_jumps_left: int = 0
var _was_on_floor: bool = false


func update(delta: float) -> void:
	_apply_gravity(delta)
	_update_timers(delta)
	_try_jump()
	_apply_jump_cut()
	_was_on_floor = body.is_on_floor()


func _apply_gravity(delta: float) -> void:
	if body.is_on_floor():
		return
	var g := gravity
	if body.velocity.y > 0.0:
		g *= fall_gravity_multiplier
	body.velocity.y = minf(body.velocity.y + g * delta, max_fall_speed)


func _update_timers(delta: float) -> void:
	if body.is_on_floor():
		_coyote = coyote_time
		_air_jumps_left = maxi(max_jumps - 1, 0)
		if not _was_on_floor:
			landed.emit()
	else:
		_coyote = maxf(_coyote - delta, 0.0)

	_buffer = maxf(_buffer - delta, 0.0)
	if Input.is_action_just_pressed(&"jump"):
		_buffer = maxf(jump_buffer_time, delta)  # garante pelo menos este frame


func _try_jump() -> void:
	if _buffer <= 0.0:
		return
	if _coyote > 0.0 or body.is_on_floor():
		_do_jump()
	elif _air_jumps_left > 0:
		_air_jumps_left -= 1
		_do_jump()


func _do_jump() -> void:
	body.velocity.y = jump_velocity
	_coyote = 0.0
	_buffer = 0.0
	jumped.emit()


func _apply_jump_cut() -> void:
	if variable_height and Input.is_action_just_released(&"jump") and body.velocity.y < 0.0:
		body.velocity.y *= jump_cut_multiplier

## Movimento HORIZONTAL. O pai precisa ser um CharacterBody2D.
## NÃ£o chama move_and_slide(): quem manda Ã© o Player.

	INSTANT,      ## velocidade vai direto ao alvo. Seco, estilo Hollow Knight.
	ACCELERATED,  ## acelera e freia. Mais "peso", estilo Dead Cells.
}

@export var mode: Mode = Mode.ACCELERATED
@export var max_speed: float = 140.0
@export_group("SÃ³ no modo ACCELERATED")

@export var friction: float = 1500.0

func update(delta: float, dir: float, speed_multiplier: float = 1.0) -> void:
	var target := dir * max_speed * speed_multiplier
	match mode:
		Mode.INSTANT:
			body.velocity.x = target
		Mode.ACCELERATED:
			var rate := acceleration if absf(dir) > 0.01 else friction
			if not body.is_on_floor():
				rate *= air_control
			body.velocity.x = move_toward(body.velocity.x, target, rate * delta)

## O AttackComponent Ã© dono do tempo e do dano do golpe.
## Este estado controla o movimento (valores no AttackData) e decide para onde voltar quando o golpe termina.

var _queued: bool = false

func enter() -> void:
	_queued = false
	player.attack.start()

func exit() -> void:
	# Se saiu no meio do golpe (ex.: tomar dano no futuro), garante a hitbox desligada.
	player.attack.cancel()

func physics_update(delta: float) -> void:
	player.apply_gravity(delta)
	player.handle_jump_buffer(delta)

	# Pular durante o golpe NÃƒO cancela o ataque: sÃ³ dÃ¡ o impulso e o golpe segue atÃ© o fim.
	if player.can_jump() and player.jump_buffered:
		player.start_jump()

	# Altura variÃ¡vel continua valendo durante o golpe (soltar o pulo corta a subida).
	player.apply_jump_cut()

	# Movimento horizontal durante o golpe: input do jogador x multiplicador do AttackData.
	# O lado do Valtair nÃ£o muda durante o golpe, para a hitbox nÃ£o virar no meio do ataque.
	var data: AttackData = player.attack.attack_data
	var input_x: float = Input.get_axis("move_left", "move_right")
	var multiplier: float = data.ground_move_multiplier if player.is_on_floor() else data.air_move_multiplier
	player.velocity.x = input_x * player.stats.move_speed * multiplier

	# BotÃ£o apertado durante o golpe: encadeia o prÃ³ximo quando este terminar.
	if Input.is_action_just_pressed("attack"):
		_queued = true

	player.attack.update(delta)

	if not player.attack.is_attacking():
		if _queued:
			_queued = false
			player.attack.start()
		else:
			# Se um pulo comeÃ§ou neste frame (velocity.y < 0), vai para Fall mesmo com o chÃ£o do frame anterior.
			var grounded: bool = player.is_on_floor() and player.velocity.y >= 0.0

	player.move_and_slide()

extends PlayerState
## Dash horizontal curto. Sem i-frames (decisÃ£o de design): sÃ³ reposiciona o Valtair.
## Cancela o golpe corpo a corpo (decisÃ£o provisÃ³ria): ao sair do estado Attack, o
## attack.gd chama cancel() no componente. A entrada neste estado Ã© feita pelo Player
## (ver Player._physics_process), por isso nenhum outro estado precisa checar o botÃ£o.

var _time_left: float = 0.0
var _direction: float = 1.0

func enter() -> void:
	# DireÃ§Ã£o: a do input, se houver; senÃ£o, a que o Valtair estÃ¡ olhando.
	var input_x: float = Input.get_axis("move_left", "move_right")
	_direction = signf(input_x) if input_x != 0.0 else float(player.facing)
	player.update_facing_direction(_direction)

	_time_left = player.stats.dash_duration
	player.dash_cooldown_left = player.stats.dash_cooldown
	if not player.is_on_floor():
		player.air_dashes_left -= 1

func physics_update(delta: float) -> void:
	player.velocity.x = _direction * (player.stats.dash_distance / player.stats.dash_duration)

	# 0 = dash reto, sem cair durante ele.
	if player.stats.dash_gravity_multiplier <= 0.0:
		player.velocity.y = 0.0
	else:
		player.apply_gravity(delta * player.stats.dash_gravity_multiplier)

	_time_left -= delta
	if _time_left <= 0.0:
		if not player.is_on_floor():
			state_machine.transition_to("fall")
		elif Input.get_axis("move_left", "move_right") != 0.0:
			state_machine.transition_to("run")
		else:
			state_machine.transition_to("idle")
		return

	player.move_and_slide()

	assert(stats != null, "ERRO: O recurso PlayerStats.tres precisa ser atribuÃ­do no Inspetor do Player!")
	state_machine.init(self)
	_connect_hud_events()


func _physics_process(delta: float) -> void:
	# Coyote time centralizado: comeÃ§a quando o chÃ£o some sem ter sido um pulo (velocity.y >= 0).
	# Assim nenhum estado precisa lembrar de chamar start_coyote_timer().
	var on_floor: bool = is_on_floor()
	if _was_on_floor and not on_floor and velocity.y >= 0.0:
		start_coyote_timer()
	_was_on_floor = on_floor

	if coyote_timer > 0.0:
		coyote_timer -= delta

	if on_floor:
		air_dashes_left = stats.max_air_dashes
	if dash_cooldown_left > 0.0:
		dash_cooldown_left -= delta

	# Dash: transiÃ§Ã£o global, vale em qualquer estado (inclusive durante o golpe, que Ã© cancelado).
	# Roda antes dos estados, entÃ£o a troca jÃ¡ vale neste mesmo frame.
	if Input.is_action_just_pressed("dodge") and can_dash():
		state_machine.transition_to("dash")

	# Ataque ranged: aÃ§Ã£o paralela, nÃ£o trava o movimento. Bloqueado durante o golpe corpo a corpo.
	if Input.is_action_just_pressed("ranged_attack") and not attack.is_attacking():
		ranged.try_fire(facing)


func apply_gravity(delta: float) -> void:
	if is_on_floor():
		return
	var g: float = stats.gravity
	if velocity.y > 0.0:
		g *= stats.fall_gravity_multiplier
	velocity.y = move_toward(velocity.y, stats.max_fall_speed, g * delta)


func can_jump() -> bool:
	return is_on_floor() or coyote_timer > 0.0


func can_dash() -> bool:
	return dash_cooldown_left <= 0.0 and (is_on_floor() or air_dashes_left > 0)


## DÃ¡ o impulso do pulo. Usado pelo estado Jump e pelo Attack (pular durante o golpe
## nÃ£o cancela o golpe), para o pulo ter uma implementaÃ§Ã£o sÃ³.
func start_jump() -> void:
	velocity.y = stats.jump_velocity
	jump_buffered = false
	coyote_timer = 0.0
	jump_cut_available = true


func start_coyote_timer() -> void:
	coyote_timer = stats.coyote_time


func handle_jump_buffer(delta: float) -> void:
	if Input.is_action_just_pressed("jump"):
		jump_buffer_timer = stats.jump_buffer
		jump_buffered = true
	elif jump_buffered:
		jump_buffer_timer -= delta
		if jump_buffer_timer <= 0.0:
			jump_buffered = false


## Altura variÃ¡vel do pulo. Chame em TODO estado em que o Valtair pode estar subindo
## (Jump, Attack, Fall), para soltar o botÃ£o cortar a subida mesmo atacando no ar.
func apply_jump_cut() -> void:
	if not jump_cut_available:
		return
	if velocity.y >= 0.0:
		jump_cut_available = false  # passou do Ã¡pice: nÃ£o hÃ¡ mais o que cortar
	elif not Input.is_action_pressed("jump"):
		velocity.y *= stats.jump_cut_multiplier
		jump_cut_available = false


func update_facing_direction(input_x: float) -> void:
	if input_x == 0.0:
		return
	var new_facing: int = 1 if input_x > 0.0 else -1
	if new_facing == facing:
		return
	facing = new_facing
	sprite.flip_h = facing < 0
	attack.set_facing(facing)  # a hitbox acompanha o lado para onde o Valtair olha


## O Player Ã© a ponte entre os seus componentes e o HUD: repassa os sinais ao EventBus.
## Os componentes sÃ£o genÃ©ricos (servem tambÃ©m a inimigos), entÃ£o nÃ£o falam com o HUD.
func _connect_hud_events() -> void:
	health.changed.connect(EventBus.player_health_changed.emit)
	ammo.changed.connect(EventBus.player_ammo_changed.emit)
	health.died.connect(EventBus.player_died.emit)
	# Valores iniciais, depois que todos os nÃ³s (inclusive o HUD) terminarem o _ready.
	_emit_hud_state.call_deferred()

	EventBus.player_health_changed.emit(health.health, health.max_health)
	EventBus.player_ammo_changed.emit(ammo.current, ammo.max_ammo)

func setup(p_id: String, p_kind: int, p_icon: Texture2D, diameter: float, pixel_art: bool, p_static: bool = false) -> void:

	icon = p_icon

	if state == State.AVAILABLE and not static_mode:
		_t += delta
		queue_redraw()

	if static_mode or state != State.AVAILABLE:
		return
	if event is InputEventMouseButton and event.pressed and event.button_index == MOUSE_BUTTON_LEFT:
		chosen.emit(self)
		accept_event()


func _draw() -> void:
	var c := size * 0.5
	var r := icon_diameter * 0.5

	if not static_mode:
		match state:
			State.AVAILABLE:
				var pulse := 0.5 + 0.5 * sin(_t * 4.0)
				if _hover:
					draw_circle(c, r + 4.0, Color(1, 1, 1, 0.15))
				draw_arc(c, r + 3.0 + pulse * 1.5, 0.0, TAU, 32, Color(1.0, 0.88, 0.45, 0.55 + 0.45 * pulse), 1.5, true)
			State.CURRENT:
				draw_arc(c, r + 3.0, 0.0, TAU, 32, Color(1.0, 0.85, 0.3, 1.0), 2.0, true)
			State.VISITED:
				draw_arc(c, r + 2.0, 0.0, TAU, 32, Color(0.55, 0.9, 0.6, 0.8), 1.0, true)

	# Fundo escuro atrÃ¡s do Ã­cone: garante leitura sobre qualquer imagem de fundo.
	draw_circle(c, r + 1.0, Color(0.04, 0.07, 0.05, 0.85))

	var tint := Color.WHITE

			State.LOCKED: tint = Color(0.45, 0.45, 0.45, 1.0)

	if icon != null:

func _to_screen(p: Vector2) -> Vector2:
	var area_pos := map_area.position * size
	var area_size := map_area.size * size
	return area_pos + p * area_size


func _center_of(id: String) -> Vector2:
	var v: MapNodeView = _views[id]
	return v.position + v.size * 0.5


func _layout() -> void:
	for id: String in _views:
		var v: MapNodeView = _views[id]
		var p: Vector2 = MapLayout.NODES[id]["pos"]
		v.position = _to_screen(p) - v.size * 0.5

		_lines.queue_redraw()


func _place_marker() -> void:
	if _marker == null:
		return
	var ms := _marker.size
	if _current_id == "":
		_marker_base = _to_screen(MapLayout.ORIGIN) - Vector2(ms.x * 0.5, ms.y)
	else:
		# Fica ao lado esquerdo do nÃ³ atual, para nÃ£o cobrir os caminhos Ã  frente.
		var v: MapNodeView = _views[_current_id]
		var c := _center_of(_current_id)
		_marker_base = Vector2(c.x - v.icon_diameter * 0.5 - ms.x - 3.0, c.y - ms.y * 0.6)
	_marker.position = _marker_base


# ---------------------------------------------------------------- linhas

func _draw_lines() -> void:
	var origin := _to_screen(MapLayout.ORIGIN)
	for id: String in MapLayout.START:
		_draw_edge("", id, origin, _center_of(id))
	for e: Array in MapLayout.EDGES:
		_draw_edge(e[0], e[1], _center_of(e[0]), _center_of(e[1]))


func _draw_edge(from_id: String, to_id: String, a: Vector2, b: Vector2) -> void:
	var th := map_theme
	if _walked.has(from_id + ">" + to_id):
		_lines.draw_line(a, b, th.line_color_walked, th.line_width + 0.5, true)
	elif _current_id == from_id and (_views[to_id] as MapNodeView).state == MapNodeView.State.AVAILABLE:
		_lines.draw_dashed_line(a, b, th.line_color_available, th.line_width + 0.5, 4.0, true, true)
	else:
		_lines.draw_dashed_line(a, b, th.line_color, th.line_width, 3.0, true, true)


# ---------------------------------------------------------------- lÃ³gica de escolha

func _on_view_chosen(view: MapNodeView) -> void:
	if view.state != MapNodeView.State.AVAILABLE:
		return

	# Os outros nÃ³s disponÃ­veis ficam bloqueados: nÃ£o dÃ¡ para voltar atrÃ¡s.
	for id: String in _views:
		var other: MapNodeView = _views[id]
		if other.state == MapNodeView.State.AVAILABLE:
			other.state = MapNodeView.State.LOCKED
	if _current_id != "":
		(_views[_current_id] as MapNodeView).state = MapNodeView.State.VISITED

	_walked[_current_id + ">" + view.id] = true
	_current_id = view.id
	view.state = MapNodeView.State.CURRENT

	var next: Array[String] = _children_of[view.id]
	for child in next:
		(_views[child] as MapNodeView).state = MapNodeView.State.AVAILABLE

	_place_marker()
	_lines.queue_redraw()
	node_chosen.emit(view.id, view.kind)
	if next.is_empty():
		route_finished.emit()

const PLACEHOLDER_SIZE: Vector2 = Vector2(6.0, 6.0)

func _ready() -> void:
	health_bar = get_node_or_null("%HealthBar") as ProgressBar
	ammo_row = get_node_or_null("%AmmoRow") as HBoxContainer
	if health_bar == null or ammo_row == null:
		_build_ui()

	EventBus.player_health_changed.connect(_on_health_changed)
	EventBus.player_ammo_changed.connect(_on_ammo_changed)


## Monta a interface por cÃ³digo (usado quando a cena nÃ£o traz os nÃ³s prontos).
func _build_ui() -> void:
	var margin: MarginContainer = MarginContainer.new()
	margin.add_theme_constant_override("margin_left", 8)
	margin.add_theme_constant_override("margin_top", 8)
	add_child(margin)

	var vbox: VBoxContainer = VBoxContainer.new()
	vbox.add_theme_constant_override("separation", 4)
	margin.add_child(vbox)

	health_bar = ProgressBar.new()
	health_bar.custom_minimum_size = Vector2(80.0, 8.0)
	health_bar.show_percentage = false
	vbox.add_child(health_bar)

	ammo_row = HBoxContainer.new()
	ammo_row.add_theme_constant_override("separation", 2)
	vbox.add_child(ammo_row)


func _on_health_changed(current: int, maximum: int) -> void:
	health_bar.max_value = maximum
	health_bar.value = current


func _on_ammo_changed(current: int, maximum: int) -> void:
	# Ajusta a quantidade de Ã­cones sÃ³ se a capacidade mudou (ex.: relÃ­quia no futuro).
	while ammo_row.get_child_count() > maximum:
		var extra: Node = ammo_row.get_child(ammo_row.get_child_count() - 1)
		ammo_row.remove_child(extra)
		extra.queue_free()
	while ammo_row.get_child_count() < maximum:
		ammo_row.add_child(_make_icon())
	for i in maximum:
		_set_icon_filled(ammo_row.get_child(i) as Control, i < current)


func _make_icon() -> Control:
	if stone_full != null:
		var texture_rect: TextureRect = TextureRect.new()
		texture_rect.stretch_mode = TextureRect.STRETCH_KEEP
		return texture_rect
	var color_rect: ColorRect = ColorRect.new()
	color_rect.custom_minimum_size = PLACEHOLDER_SIZE
	return color_rect


func _set_icon_filled(icon: Control, filled: bool) -> void:
	if icon is TextureRect:
		(icon as TextureRect).texture = stone_full if filled else stone_empty
	elif icon is ColorRect:
		(icon as ColorRect).color = COLOR_FULL if filled else COLOR_EMPTY
